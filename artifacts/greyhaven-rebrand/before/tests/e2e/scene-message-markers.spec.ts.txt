import { expect, test } from '@playwright/test'
import { ActIFinalFixture } from '../helpers/actIFinal'
import { openSophie, seedScene } from './scene01-helper'

test('FINAL FIND DANIEL keeps one Sophie scene marker across actors, Send, typing, reopen and reload', async ({ page }) => {
  const fixture = new ActIFinalFixture().until('OPEN-S02-C03')
  const savedLines = fixture.save.prologue!.scene01!.groupMessages
  await seedScene(page, fixture.save)
  await page.getByRole('button', { name: 'Open Messages', exact: true }).first().click()
  await page.getByRole('button', { name: 'Open FIND DANIEL group conversation', exact: true }).click()
  const markers = page.locator('.scene01-chat .new-message-marker')
  await expect(markers).toHaveCount(1)
  await expect(markers).toHaveText('NEW MESSAGE · Sophie')
  const placement = await markers.getAttribute('data-message-id')
  const bubbles = page.locator('.opening-thread .opening-bubble')
  const expected = savedLines.filter(line => line.sender !== 'system').map(line => line.text)
  await expect(bubbles).toHaveText(expected)
  await page.locator('.scene01-options .opening-choice').first().click()
  await expect(bubbles).toHaveText(expected)
  await expect(markers).toHaveCount(1)
  await page.getByRole('button', { name: 'Send', exact: true }).click()
  await page.clock.runFor(1000)
  await expect(markers).toHaveCount(1)
  await page.clock.runFor(120000)
  const history = await bubbles.allTextContents()
  await page.getByRole('button', { name: 'Back to inbox' }).click()
  await page.getByRole('button', { name: 'Open FIND DANIEL group conversation', exact: true }).click()
  await expect(markers).toHaveCount(1)
  await expect(markers).toHaveAttribute('data-message-id', placement!)
  await page.reload()
  await page.getByRole('button', { name: 'Enter the archive', exact: true }).click()
  await page.getByRole('button', { name: 'Open Messages', exact: true }).first().click()
  await page.getByRole('button', { name: 'Open FIND DANIEL group conversation', exact: true }).click()
  await expect(markers).toHaveCount(1)
  await expect(markers).toHaveAttribute('data-message-id', placement!)
  await expect(bubbles).toHaveText(history)
  await markers.scrollIntoViewIfNeeded()
  await expect(markers).toBeInViewport()
  await page.screenshot({ path: 'artifacts/scene-message-markers/find-daniel.png' })
})

test('Sophie history retains separate markers for first contact and the actual private scene', async ({ page }) => {
  await seedScene(page, new ActIFinalFixture().until('OPEN-S03-C02').save)
  await openSophie(page)
  const markers = page.locator('.scene01-chat .new-message-marker')
  await expect(markers).toHaveCount(2)
  await expect(markers.nth(0)).toHaveAttribute('data-scene-id', 'OPEN-S01')
  await expect(markers.nth(1)).toHaveAttribute('data-scene-id', 'ch01_sc03_just_between_us')
  await expect(markers.nth(1)).toHaveAttribute('data-message-id', 'OPEN-S03-FIRST-MESSAGE')
  const placements = await markers.evaluateAll(elements => elements.map(el => el.getAttribute('data-message-id')))
  await page.getByRole('button', { name: 'Back to inbox' }).click()
  await page.getByRole('button', { name: 'Open Sophie conversation', exact: true }).click()
  expect(await markers.evaluateAll(elements => elements.map(el => el.getAttribute('data-message-id')))).toEqual(placements)
})
