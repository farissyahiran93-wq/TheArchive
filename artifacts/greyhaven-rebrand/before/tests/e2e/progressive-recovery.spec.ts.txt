import { expect, test } from '@playwright/test'
import { completeOpening } from './opening-helper'

test('partial snapshot distinguishes readable fragments, passwords and unavailable keys', async ({ page }) => {
  await page.goto('/?legacyopening=1')
  await page.getByRole('button', { name: "Open Sophie's new message" }).click()
  await page.getByLabel('Your name').fill('Rae')
  await page.getByRole('button', { name: 'Set up phone' }).click()
  await completeOpening(page)
  await page.getByRole('button', { name: 'Open The Archive', exact: true }).click()
  await page.getByRole('button', { name: 'Settings', exact: true }).click()
  await expect(page.getByText('Partial', { exact: true })).toBeVisible()
  await expect(page.getByText('Unverified', { exact: true })).toBeVisible()
  await page.getByRole('button', { name: "Daniel's Phone", exact: true }).click()
  await page.locator('.mirror-icon').filter({ hasText: 'Maps' }).click()
  await page.getByRole('button', { name: 'Route preview · unavailable', exact: true }).click()
  await expect(page.getByText(/A route cache exists/)).toBeVisible()
  await expect(page.getByLabel('Password', { exact: true })).toHaveCount(0)
  await page.getByRole('button', { name: "Daniel's Phone", exact: true }).click()
  await page.locator('.mirror-icon').filter({ hasText: 'Files' }).click()
  await page.getByRole('button', { name: 'protected/container.enc', exact: true }).click()
  await expect(page.getByText(/The key and original contents were not recovered/)).toBeVisible()
  await expect(page.getByLabel('Password', { exact: true })).toHaveCount(0)
  await page.getByRole('button', { name: "Daniel's Phone", exact: true }).click()
  await page.locator('.mirror-icon').filter({ hasText: 'Notes' }).click()
  await page.getByRole('button', { name: 'Locked Note — public comparison', exact: true }).click()
  await page.getByLabel('Password', { exact: true }).fill('BUS4')
  await page.getByRole('button', { name: 'Unlock', exact: true }).click()
  await expect(page.getByText(/Compare their visible Escort fields yourself/)).toBeVisible()
  await expect(page.locator('.mirror-document')).not.toContainText('June Ellis')
  await page.screenshot({ path: test.info().outputPath('public-comparison.png'), animations: 'disabled' })
})
