import { expect,test,type Page } from '@playwright/test'
import { Chapter1Fixture } from '../helpers/chapter1'
import { sceneAt,seedScene } from './scene01-helper'

async function click(page:Page,name:string){await page.getByRole('button',{name,exact:true}).first().click()}
async function inspect(page:Page,title:string){await page.getByRole('button',{name:new RegExp('^'+title.replace(/[.*+?^${}()|[\]\\]/g,'\\$&'))}).first().click();await expect(page.getByRole('article',{name:'Investigation record'})).toBeVisible();await click(page,'Back to investigation')}
async function advance(page:Page,n:number){await click(page,'Continue investigation');await expect(page.getByRole('region',{name:'Chapter 1 investigation'})).toHaveAttribute('data-chapter1-scene',String(n))}
async function send(page:Page,text:string){await click(page,text);await click(page,'Send');await page.clock.runFor(3000)}
async function choose(page:Page,key:string,id:string){await page.getByRole('combobox',{name:`Interpret ${key}`}).selectOption(id);await click(page,'Record interpretation')}

test('official structure: manual sources, diff, contacts, count, claim, board and role end Chapter 1 without launching Chapter 2',async({page})=>{
  test.setTimeout(180000)
  await page.setViewportSize({width:320,height:720})
  await seedScene(page,new Chapter1Fixture().opening().save)
  await click(page,'Open The Archive')
  await expect(page.getByRole('region',{name:'Chapter 1 investigation'})).toHaveAttribute('data-chapter1-scene','4')
  await expect(page.getByText('Post-verification statement — partial reconstruction',{exact:true})).toHaveCount(0)
  await expect(page.getByRole('button',{name:'Activate recovered reference'})).toBeDisabled()
  await inspect(page,'Daniel Mercer — public report')
  await click(page,'Activate recovered reference')
  await inspect(page,'Recovered publication reference')
  await page.reload();await click(page,'Enter the archive');await click(page,'Open The Archive')
  await advance(page,5)
  await inspect(page,'Earlier recovered report layer');await click(page,'Open version comparison')
  await expect(page.getByLabel('Version comparison')).toContainText('JUNE ELLIS')
  await click(page,'Inspect and save comparison');await advance(page,6)
  await click(page,'Examine changed Escort field');await choose(page,'missing','insufficient');await advance(page,7)
  await click(page,'Open Nora reporting inbox');await send(page,'I only want to discuss the public record for now.');await click(page,'Back from conversation');await advance(page,8)
  await click(page,'Ask Nora about her retained page');await send(page,'Can I examine the page you kept?');await click(page,'Back from conversation')
  await inspect(page,'Page retained by Nora');await click(page,'Compare retained page with both source layers');await click(page,'Record retained-page comparison');await advance(page,9)
  await inspect(page,'Source trace — incomplete');await click(page,'Record source uncertainty; sender identity remains unknown');await advance(page,10)
  await click(page,'Authorize limited snapshot');await inspect(page,'To myself — recovered fragment');await advance(page,11)
  await inspect(page,'Archive Index — integrity report');await inspect(page,'Index caution — ordinary recovered record')
  await choose(page,'index','equal');await expect(page.getByRole('button',{name:'Continue investigation'})).toBeDisabled()
  await choose(page,'index','14-13')
  await page.screenshot({path:test.info().outputPath('scene11-count.png')})
  await page.reload();await click(page,'Enter the archive');await click(page,'Open The Archive');await advance(page,12)
  await send(page,'Sophie, could you put me in touch with Eli for a general recovery question?')
  await click(page,'Open Eli conversation');await send(page,'Please pass on a general question about incomplete recovery.');await click(page,'Back from conversation')
  await click(page,'Acknowledge recovery limitations');await advance(page,13)
  await click(page,'Open Maya conversation');await send(page,'We can keep personal details private.');await click(page,'Back from conversation');await advance(page,14)
  await click(page,'Open Lucas conversation');await send(page,'We can leave this here if you do not want to talk.');await click(page,'Back from conversation');await advance(page,15)
  await inspect(page,'Post-verification statement — partial reconstruction');await click(page,'Examine “after verification” against the index');await choose(page,'claim','unverified');await advance(page,16)
  await send(page,'Nora, can you provide the earliest copy of the public linked record? I am not attaching private recovered files.');await advance(page,17)
  for(const [key,id] of [['diff','observed'],['claim','reported'],['source','unknown'],['theory','interpretation']])await page.getByRole('combobox',{name:`Classify ${key}`}).selectOption(id)
  await click(page,'Check classifications');await click(page,'Save conclusion: versions differ; source and actor remain unproven');await advance(page,18)
  await inspect(page,'Correction workflow — status trace');await choose(page,'ending','cautious')
  await page.reload();await click(page,'Enter the archive');await click(page,'Open The Archive')
  await click(page,'End Chapter')
  await expect(page.getByRole('heading',{name:'CHAPTER 1 COMPLETE'})).toBeVisible()
  await expect(page.getByRole('button',{name:'Continue to Chapter 2'})).toBeVisible()
  await page.screenshot({path:test.info().outputPath('chapter-complete.png')})
  await page.clock.runFor(20000)
  const pending=await page.evaluate(()=>sessionStorage.getItem('the-archive-reboot:pending-save:v1'))
  if(pending)expect(JSON.parse(pending).save.chapter2Access).toBeUndefined()
  await click(page,'Continue to Chapter 2')
  await expect(page.getByRole('region',{name:'The Archive access extension'})).toBeVisible()
  await click(page,'Back to home');await click(page,'Open Messages')
  await expect(page.getByRole('button',{name:'Open Chapter 1 nora conversation'})).toBeVisible()
  await click(page,'Open Chapter 1 nora conversation')
  await expect(page.getByRole('region',{name:'Nora Chapter 1 conversation'})).toContainText("I've recorded your request.")
})

test('Unknown warning is once-only and leaves Lucas choice intact',async({page})=>{
  await seedScene(page,sceneAt('G01'))
  await click(page,'Open Messages');await click(page,'Open FIND DANIEL group conversation')
  const group=page.getByRole('region',{name:'FIND DANIEL group conversation'})
  await expect(group).toHaveAttribute('data-scene-node','OPEN-S02-C01')
  // Waking the app is a scheduler tick; the persisted group pause remains unchanged.
  await page.evaluate(()=>document.dispatchEvent(new Event('visibilitychange')))
  await page.clock.runFor(2000)
  await expect(page.getByLabel('Incoming notification',{exact:true})).toContainText('Unknown Sender')
  await expect(group).toHaveAttribute('data-scene-node','OPEN-S02-C01')
  await click(page,'Open notification from Unknown Sender')
  await click(page,'Read warning and show responses');await click(page,'Who is this?')
  await expect(page.locator('.c1-contact .opening-bubble.player')).toHaveCount(0)
  await click(page,'Send');await expect(page.locator('.c1-contact .opening-bubble.player')).toHaveCount(1)
  await page.reload();await click(page,'Enter the archive');await click(page,'Open Messages');await click(page,'Open Chapter 1 Unknown Sender conversation')
  await expect(page.getByText('Be careful who you trust in that group.',{exact:true})).toHaveCount(1)
})

test('warning disclosure is optional, selected before Send and acknowledged by Sophie only afterwards',async({page})=>{
  await seedScene(page,new Chapter1Fixture().opening().act({type:'read-sender'}).save)
  await click(page,'Open Messages');await click(page,'Open Sophie conversation')
  await page.getByText('Optional: tell Sophie about the warning',{exact:true}).click()
  const text='An unknown sender warned me to be careful who I trust in the group.'
  await click(page,text)
  await expect(page.locator('.opening-bubble.player').filter({hasText:text})).toHaveCount(0)
  await click(page,'Send')
  await expect(page.locator('.opening-bubble.player').filter({hasText:text})).toHaveCount(1)
  await page.clock.runFor(2000)
  await expect(page.getByRole('region',{name:'Sophie private conversation'}).getByText("Thank you for telling me. I don't know who sent that.",{exact:true})).toBeVisible()
})

test('no-sharing technical guidance is an explicit action and does not send a player message to Eli',async({page})=>{
  const f=new Chapter1Fixture().opening().run(11).send('eli-intro','introduce')
  await seedScene(page,f.save)
  await click(page,'Open The Archive');await click(page,'Open Eli conversation')
  await click(page,'I would rather read a general technical explanation without sharing my files.')

  await click(page,'Read guidance without sharing')
  await expect(page.locator('.c1-contact .opening-bubble.player')).toHaveCount(0)
  await click(page,'Back from conversation')
  await expect(page.getByText(/General recovery guidance:/)).toBeVisible()
  await click(page,'Acknowledge recovery limitations');await advance(page,13)
})
